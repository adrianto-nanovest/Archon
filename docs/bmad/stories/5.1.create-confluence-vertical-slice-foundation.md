    ```typescript
    export function useCreateSource() {
      const queryClient = useQueryClient();
      const { showToast } = useToast();

      return useMutation({
        mutationFn: (request: CreateSourceRequest) => confluenceService.createSource(request),
        onMutate: async (newSource) => {
          await queryClient.cancelQueries({ queryKey: confluenceKeys.lists() });
          const previous = queryClient.getQueryData<ConfluenceSourceListResponse>(confluenceKeys.lists());

          const optimisticSource = createOptimisticEntity<ConfluenceSource>({
            source_type: "confluence",
            space_key: newSource.space_key,
            base_url: newSource.base_url,
            status: "ready",
            last_sync: null,
            total_pages: 0,
            created_at: new Date().toISOString(),
            updated_at: new Date().toISOString(),
          });

          queryClient.setQueryData<ConfluenceSourceListResponse>(confluenceKeys.lists(), (old) => ({
            sources: [...(old?.sources || []), optimisticSource],
            count: (old?.count || 0) + 1,
          }));

          return { previous, localId: optimisticSource._localId };
        },
        onError: (err, variables, context) => {
          if (context?.previous) {
            queryClient.setQueryData(confluenceKeys.lists(), context.previous);
          }
          showToast(err instanceof Error ? err.message : "Failed to create source", "error");
        },
        onSuccess: (data) => {
          queryClient.invalidateQueries({ queryKey: confluenceKeys.lists() });
          showToast(`Confluence source created: ${data.space_key}`, "success");
        },
      });
    }
    ```

    ```typescript
    export function useDeleteSource() {
      const queryClient = useQueryClient();
      const { showToast } = useToast();

      return useMutation({
        mutationFn: (sourceId: string) => confluenceService.deleteSource(sourceId),
        onMutate: async (sourceId) => {
          await queryClient.cancelQueries({ queryKey: confluenceKeys.lists() });
          const previous = queryClient.getQueryData<ConfluenceSourceListResponse>(confluenceKeys.lists());

          queryClient.setQueryData<ConfluenceSourceListResponse>(confluenceKeys.lists(), (old) => ({
            sources: (old?.sources || []).filter(s => s.source_id !== sourceId),
            count: Math.max(0, (old?.count || 0) - 1),
          }));

          return { previous };
        },
        onError: (err, sourceId, context) => {
          if (context?.previous) {
            queryClient.setQueryData(confluenceKeys.lists(), context.previous);
          }
          showToast(err instanceof Error ? err.message : "Failed to delete source", "error");
        },
        onSuccess: () => {
          queryClient.invalidateQueries({ queryKey: confluenceKeys.all });
          showToast("Source deleted successfully", "success");
        },
      });
    }
    ```

### Architecture Context

#### Vertical Slice Pattern

**Source:** [docs/bmad/brownfield-architecture/source-tree-and-module-organization.md](../../brownfield-architecture/source-tree-and-module-organization.md)

The Confluence feature follows the established vertical slice architecture:

```text
archon-ui-main/src/features/confluence/
├── components/           # UI components (Story 5.2)
├── hooks/
│   └── useConfluenceQueries.ts  # Query hooks & keys
├── services/
│   └── confluenceService.ts     # API calls
├── types/
│   └── index.ts                 # TypeScript types
└── index.ts                     # Barrel export
```

**Pattern Reference:** See `features/knowledge/` and `features/projects/` for complete examples.

---

#### Query Key Factory Pattern

**Source:** [PRPs/ai_docs/QUERY_PATTERNS.md](../../../../PRPs/ai_docs/QUERY_PATTERNS.md)

Every feature MUST implement a query key factory:

```typescript
export const confluenceKeys = {
  all: ["confluence"] as const,                                    // Base key for the domain
  lists: () => [...confluenceKeys.all, "list"] as const,          // For list endpoints
  detail: (id: string) => [...confluenceKeys.all, "detail", id] as const, // For single item
  status: (id: string) => [...confluenceKeys.all, "status", id] as const, // For sync status
};
```

**CRITICAL:** Query keys must follow the factory pattern from `shared/config/queryPatterns.ts`:
- Always use `DISABLED_QUERY_KEY` when query should not execute
- Always use `STALE_TIMES` constants for staleTime configuration
- Never hardcode stale times directly in hooks

---

#### Shared Patterns Import

**Source:** [archon-ui-main/src/features/shared/config/queryPatterns.ts](../../../../archon-ui-main/src/features/shared/config/queryPatterns.ts)

```typescript
import { DISABLED_QUERY_KEY, STALE_TIMES } from "@/features/shared/config/queryPatterns";
```

**STALE_TIMES Reference:**
- `instant: 0` - Always fresh (real-time data)
- `realtime: 3_000` - 3 seconds (near real-time)
- `frequent: 5_000` - 5 seconds (sync status polling)
- `normal: 30_000` - 30 seconds (standard cache for lists)
- `rare: 300_000` - 5 minutes (rarely changing config)
- `static: Infinity` - Never stale (settings)

#### API Client Usage

**Source:** [archon-ui-main/src/features/shared/api/apiClient.ts](../../../../archon-ui-main/src/features/shared/api/apiClient.ts)

All service methods MUST use `callAPIWithETag<T>()`:

```typescript
import { callAPIWithETag } from "../../shared/api/apiClient";

// GET request
async listSources(): Promise<ConfluenceSourceListResponse> {
  return callAPIWithETag<ConfluenceSourceListResponse>("/api/confluence/sources");
}

// POST request with body
async createSource(request: CreateSourceRequest): Promise<ConfluenceSource> {
  return callAPIWithETag<ConfluenceSource>("/api/confluence/sources", {
    method: "POST",
    body: JSON.stringify(request),
  });
}

// DELETE request
async deleteSource(sourceId: string): Promise<DeleteSourceResponse> {
  return callAPIWithETag<DeleteSourceResponse>(`/api/confluence/${sourceId}`, {
    method: "DELETE",
  });
}
```

**Browser ETag Handling:** The browser automatically handles ETags and 304 responses. No manual ETag management needed.
