#### Backend Search API Already Supports Confluence Filters

**Source:** [python/src/server/api_routes/knowledge_api.py](../../../../python/src/server/api_routes/knowledge_api.py#L177-L214)

The `RagQueryRequest` model already includes Confluence-specific filter parameters:

```python
class RagQueryRequest(BaseModel):
    query: str
    source: str | None = None
    match_count: int = 5
    return_mode: str = "chunks"  # "chunks" or "pages"
    # Confluence-specific filters
    space_key: str | None = None  # Filter by Confluence space key
    jira_issue: str | None = None  # Filter by linked JIRA issue key
    hierarchy_path: str | None = None  # Filter by page hierarchy path prefix
    mentioned_user: str | None = None  # Filter by mentioned user account_id
    # Pagination support (Story 4.3)
    offset: int = Field(default=0, ge=0)
    limit: int = Field(default=50, ge=1, le=100)
```

**API Endpoint:** `POST /api/knowledge-items/search` (or `/api/rag/query`)

#### URL Query Param Persistence Pattern

**Standard Pattern for Filter Persistence:**

```typescript
// Read from URL on mount
const searchParams = new URLSearchParams(window.location.search);
const initialFilters = {
  sourceType: searchParams.get("source") || "all",
  spaceKeys: searchParams.get("spaces")?.split(",").filter(Boolean) || [],
  hasJiraLinks: searchParams.get("jira") === "true",
};

// Update URL on filter change (without page reload)
const updateUrlParams = (filters: SearchFilters) => {
  const params = new URLSearchParams();
  if (filters.sourceType !== "all") params.set("source", filters.sourceType);
  if (filters.spaceKeys.length > 0) params.set("spaces", filters.spaceKeys.join(","));
  if (filters.hasJiraLinks) params.set("jira", "true");

  const newUrl = params.toString() ? `?${params.toString()}` : window.location.pathname;
  window.history.replaceState({}, "", newUrl);
};
```

#### JIRA Integration Pattern

```typescript
interface JiraChipProps {

    >
      {issueKey}
    </button>
  );
};
```

#### Breadcrumb Truncation Logic

**Truncation at 5 levels with "...":**

```typescript
const renderBreadcrumbs = (path: string | null) => {
  if (!path) return null;

  // path format: "/parent_id/child_id/grandchild_id"
  const segments = path.split("/").filter(Boolean);

  if (segments.length === 0) return null;

  // Truncate if more than 5 levels
  const displaySegments = segments.length > 5
    ? [...segments.slice(0, 2), "...", ...segments.slice(-2)]
    : segments;

  return (
    <nav aria-label="Page hierarchy" className="flex items-center gap-1 text-xs text-gray-400">
      {displaySegments.map((segment, index) => (
        <React.Fragment key={index}>
          {index > 0 && <span className="mx-1">/</span>}
          <span className={segment === "..." ? "text-gray-500" : ""}>{segment}</span>
        </React.Fragment>
      ))}
    </nav>
  );
};
```
