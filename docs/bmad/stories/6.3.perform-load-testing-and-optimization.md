### Memory Profiling Pattern

```python
import tracemalloc
import psutil

def measure_sync_memory():
    process = psutil.Process()
    initial_memory = process.memory_info().rss / 1024 / 1024  # MB

    tracemalloc.start()

    # Run sync operation
    await sync_service.sync_space(...)

    current, peak = tracemalloc.get_traced_memory()
    tracemalloc.stop()

    final_memory = process.memory_info().rss / 1024 / 1024  # MB
    memory_increase = ((final_memory - initial_memory) / initial_memory) * 100

    assert memory_increase < 20, f"Memory increased by {memory_increase:.1f}% (target: <20%)"
```

### Latency Percentile Calculation

```python
import numpy as np
import statistics

def calculate_latency_percentiles(latencies: list[float]) -> dict:
    """Calculate p50, p95, p99 latencies."""
    return {
        "p50": np.percentile(latencies, 50),
        "p95": np.percentile(latencies, 95),
        "p99": np.percentile(latencies, 99),
        "mean": statistics.mean(latencies),
        "stdev": statistics.stdev(latencies) if len(latencies) > 1 else 0,
    }
```

### EXPLAIN ANALYZE Query Pattern

```python
async def verify_index_usage(supabase_client, query: str) -> bool:
    """Verify that search query uses indexes (no sequential scans)."""
    result = supabase_client.rpc(
        "explain_analyze_search",
        {"query_text": query}
    ).execute()

    plan = result.data[0]["QUERY PLAN"] if result.data else ""

    # Assert no sequential scans on large tables
    assert "Seq Scan on archon_crawled_pages" not in plan
    assert "Index Scan" in plan or "Bitmap Index Scan" in plan

    return True
```
